# test262 exclusions for the SimpleJS conformance harness.
#
# An exclusion says "we deliberately do not measure this", so every line carries a reason and is
# counted and reported by category. This file is the machine-readable form of the *Deliberately
# unimplemented ES2026 features* section of docs/simplejs.md — keep the two in step.
#
# Line kinds:
#   dir:     <path prefix relative to test262/test/>   # reason
#   keep:    <path prefix re-admitted from a dir:>     # reason
#   feature: <features: value>                         # reason
#   pattern: <python regex matched against the source> # reason
#
# A `keep:` line wins over any `dir:` line, and exists so a broad directory exclusion cannot
# silently swallow a subtree that is failing for a reason we own. It overrides `dir:` only —
# `feature:` and `pattern:` still apply.
#
# A `pattern:` line is a source scan, not a feature flag, because test262 does not flag `eval` or the
# Function constructor. It can false-positive on a test that merely mentions the word; the count is
# reported so the cost stays visible. The alternative is thousands of guaranteed failures carrying no
# information.
#
# Deliberately NOT excluded, even though it would be tempting: resizable-arraybuffer (supported,
# including length-tracking views) and explicit-resource-management (supported). Anything the engine
# gets wrong there belongs in config/test262-baseline.txt, not here.

dir:     intl402/                    # Intl not implemented (deliberate)
dir:     staging/                    # not part of the conformance corpus
dir:     annexB/                     # normative-optional web-legacy semantics; not required of a non-browser host
dir:     built-ins/RegExp/property-escapes/generated/  # exhaustive per-code-point assertions against one UCD version
keep:    built-ins/RegExp/property-escapes/generated/strings/  # properties of strings: our own gap, not the JDK's
dir:     language/eval-code/indirect/  # indirect eval — written (0, eval)(…) or through an alias, so no source pattern catches it
dir:     built-ins/eval/               # the descriptor surface of a global eval that deliberately does not exist
#
# The property-escapes pair is not a feature gap. With no ICU dependency our Unicode data is
# whatever the JDK ships, and the project builds on both 25 (Unicode 16.0) and 26 (Unicode 17.0),
# so those generated files disagree across supported build JDKs (new scripts such as Sidetic and
# Tolong_Siki do not even compile on 25) and no single baseline can be green on both. Measured
# either way: the hand-written property-escapes tests beside them (grammar-extension,
# loose-matching, unsupported-property, non-existent-property), which exercise RegexTranslator's
# own grammar and validation, and the generated/strings/ subtree kept above — those fail because
# properties of strings (\p{RGI_Emoji} and friends) are unimplemented, which is our gap to own,
# and they answer identically on both JDKs since the translator rejects the property before
# java.util.regex ever sees it.

feature: Temporal                    # deliberate: Date is the only temporal type
feature: SharedArrayBuffer           # deliberate: single-threaded per-request VM
feature: Atomics                     # deliberate: ditto
feature: Atomics.pause               # deliberate: ditto
feature: WeakRef                     # deliberate: GC-observable behaviour
feature: FinalizationRegistry        # deliberate: ditto
feature: Symbol.species              # deliberate: JsArray/JsTypedArray are not subclassable
feature: tail-call-optimization      # deliberate: no TCO
feature: cross-realm                 # needs $262.createRealm, which the shim does not provide
feature: IsHTMLDDA                   # host-specific exotic object
feature: immutable-arraybuffer       # deliberate: not in the ES2026 snapshot, no third buffer state
feature: dynamic-import              # deliberate: import() resolves only the args/db host built-ins
feature: import-assertions           # deliberate: arbitrary module resolution is out of scope
feature: import-attributes           # deliberate: ditto
feature: json-modules                # deliberate: ditto
feature: import-defer                # deliberate: ditto
feature: source-phase-imports        # deliberate: ditto
feature: arbitrary-module-namespace-names  # deliberate: ditto
feature: ShadowRealm                 # deliberate: evaluate(sourceText) is runtime codegen from a string, like eval

pattern: \beval\s*\(                 # eval absent by design (sandbox boundary)
pattern: \bFunction\s*\(             # Function constructor disabled by design (called or newed)
pattern: \bnew\s+Function\s*;        # ditto, the bare `new Function;` form. Deliberately not the broader
                                     # `\bnew\s+Function\b`, which also swallowed `new Function.prototype.apply()`
                                     # — those measure [[Construct]], not the constructor we do not provide
pattern: \$262\.(createRealm|evalScript|agent|gc|IsHTMLDDA)  # host APIs the shim does not provide
